#include <Arduino.h>
#include <HardwareSerial.h>
#include <TinyGPSPlus.h>
#include <WiFi.h>
#include <HTTPClient.h>

// --- HARDWARE SERIAL PORT FOR GPS ---
// Port 1 for GPS (UART1)
HardwareSerial gpsSerial(1); 
const int GPS_RX_PIN = 16;
const int GPS_TX_PIN = 17;

// --- WIFI CONFIGURATION ---
const char* ssid = "khalid";
const char* password = "AA04C8D5";

// --- TRACCAR CONFIGURATION ---
const char server[] = "81.10.85.12";   // Your Traccar IP
const int port = 5055;                // Traccar OsMand Port
const char deviceID[] = "nano_tracker"; // Set to your preferred identifier

// --- OBJECTS & TIMERS ---
TinyGPSPlus gps;
unsigned long lastSend = 0;
const unsigned long sendInterval = 10000; // Send every 10 seconds

/**
 * @brief Connects to WiFi network
 */
void connectWiFi() {
  Serial.println();
  Serial.println("Connecting to WiFi...");
  
  WiFi.begin(ssid, password);
  
  // Wait until connected
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println();
  Serial.println("WiFi connected!");
  Serial.print("IP address: ");
  Serial.println(WiFi.localIP());
}

/**
 * @brief Sends GPS data to Traccar server via HTTP GET
 */
void sendDataToTraccar(double lat, double lon, double speed, double alt) {
  Serial.println(F("\n--- Starting Traccar Transmission ---"));

  // 1. Check WiFi Connection
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi disconnected. Reconnecting...");
    connectWiFi();
  }

  // 2. Construct Traccar URL (OsMand Protocol)
  String url = "http://" + String(server) + ":" + String(port) + "/";
  url += "?id=" + String(deviceID);
  url += "&lat=" + String(lat, 6);
  url += "&lon=" + String(lon, 6);
  url += "&speed=" + String(speed, 2);
  url += "&alt=" + String(alt, 2);

  Serial.print(F("URL: "));
  Serial.println(url);

  // 3. Send HTTP GET Request
  HTTPClient http;
  http.begin(url);
  
  int httpResponseCode = http.GET();
  
  if (httpResponseCode > 0) {
    Serial.print(F("HTTP Response code: "));
    Serial.println(httpResponseCode);
    
    if (httpResponseCode == 200) {
      Serial.println(F("✅ Success! Data sent to Traccar."));
      String response = http.getString();
      Serial.print(F("Response: "));
      Serial.println(response);
    } else {
      Serial.println(F("⚠️ Server returned non-200 status."));
    }
  } else {
    Serial.print(F("❌ HTTP Request failed. Error: "));
    Serial.println(http.errorToString(httpResponseCode));
  }
  
  http.end();
}

void setup() {
  // 1. Setup USB Serial for debugging
  Serial.begin(115200); 
  delay(1000);
  Serial.println("\n=================================");
  Serial.println("ESP32-S3 WiFi GPS Tracker");
  Serial.println("=================================");

  // 2. Setup GPS on UART1
  Serial.println("Initializing GPS...");
  gpsSerial.begin(9600, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
  Serial.println("GPS initialized on UART1");

  // 3. Connect to WiFi
  connectWiFi();
  
  Serial.println("\nSetup Complete. Starting loop...");
  Serial.println("=================================\n");
}

void loop() {
  // 1. Process GPS Data
  while (gpsSerial.available() > 0) {
    gps.encode(gpsSerial.read());
  }
  
  // 2. Timer Check: Is it time to send?
  if (millis() - lastSend >= sendInterval) {
    lastSend = millis();
    
    // 3. Display GPS Status
    Serial.println(F("\n--- GPS Status ---"));
    Serial.print(F("Satellites: "));
    Serial.println(gps.satellites.value());
    Serial.print(F("HDOP: "));
    Serial.println(gps.hdop.hdop());
    
    // 4. Collect GPS or Placeholder Data
    double lat = 0.0;
    double lon = 0.0;
    double speed = 0.0;
    double alt = 0.0;
    
    if (gps.location.isValid()) {
      // ✅ Use real GPS data
      lat = gps.location.lat();
      lon = gps.location.lng();
      speed = gps.speed.kmph();
      alt = gps.altitude.meters();
      
      Serial.println(F("✅ GPS FIX ACQUIRED!"));
      Serial.print(F("Latitude: ")); Serial.println(lat, 6);
      Serial.print(F("Longitude: ")); Serial.println(lon, 6);
      Serial.print(F("Speed: ")); Serial.print(speed); Serial.println(F(" km/h"));
      Serial.print(F("Altitude: ")); Serial.print(alt); Serial.println(F(" m"));
      
    } else {
      // ⚠️ Use placeholder data (0, 0)
      Serial.println(F("🔴 NO GPS FIX - Sending placeholder data"));
      if (gps.satellites.value() == 0) {
        Serial.println(F("⚠️ No satellites visible. Check GPS antenna and view of sky."));
      }
    }
    
    // 5. Send data to Traccar
    sendDataToTraccar(lat, lon, speed, alt);
    
    // 6. Display WiFi status
    Serial.print(F("WiFi Signal: "));
    Serial.print(WiFi.RSSI());
    Serial.println(F(" dBm"));
  }
}
